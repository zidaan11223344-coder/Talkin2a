# Talkin2a - Altare Environment Variables
# انسخ كل سطر إلى Environment Variables في Altare.
# استبدل القيم بين <...> فقط.
# لا تضع GITHUB_TOKEN أو كلمات مرور قاعدة البيانات داخل Git.

# حساب البوت
BOT_ID=<اسم_البوت>
BOT_PWD=<كلمة_مرور_البوت>
BOT_MASTER=<اسم_الماستر>
GROUP_TO_JOIN=<اسم_الغرفة_الأولى>

# التخزين الدائم داخل Volume مربوط على /data
BOT_DATA_DIR=/data/talkin2
PERSISTENT_DATA_DIR=/data/talkin2
TELEGRAM_LOG_FILE=/data/talkin2/bot_runtime.log
TELEGRAM_CHAT_FILE=/data/talkin2/telegram_chat_id.json

# منع دخول كل الغرف تلقائيًا عند التشغيل أو إعادة الاتصال
STARTUP_JOIN_ENABLED=0
RESTORE_SAVED_ROOMS_ON_STARTUP=0

# إظهار حالة المصادقة وWebSocket وقاعدة البيانات في Runtime Logs
QUIET_MODE=0
DEBUG=0

# قاعدة Talkin ID Data - اختر طريقة واحدة فقط
# الطريقة الأولى: رابط اتصال كامل
TALKIN_ID_DATA_DATABASE_URL=<رابط_قاعدة_البيانات>

# الطريقة الثانية: حقول منفصلة، واحذف/اترك الرابط الكامل فارغًا
TALKIN_ID_DATA_DATABASE_KIND=<mysql_أو_postgres>
TALKIN_ID_DATA_DB_HOST=<عنوان_خادم_قاعدة_البيانات>
TALKIN_ID_DATA_DB_PORT=<3306_لـ_MySQL_أو_5432_لـ_PostgreSQL>
TALKIN_ID_DATA_DB_NAME=<اسم_قاعدة_البيانات>
TALKIN_ID_DATA_DB_USER=<مستخدم_قاعدة_البيانات>
TALKIN_ID_DATA_DB_PASSWORD=<كلمة_مرور_قاعدة_البيانات>

# حفظ ملفات الحالة JSON في مستودع بيانات Talkin-ID
GITHUB_SYNC=1
GITHUB_REPO=zidaan11223344-coder/Talkin-ID
GITHUB_BRANCH=main
GITHUB_DATA_DIR=bot_data
GITHUB_TOKEN=<GitHub_Token_بصلاحية_Contents_Read_and_Write_إلى_Talkin-ID>

GITHUB_JSON_MIRROR=0

# اختياري: Telegram لإرسال السجل والمتغيرات
